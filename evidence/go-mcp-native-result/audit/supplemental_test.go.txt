package mcp

import (
	"context"
	"encoding/json"
	"testing"

	"github.com/modelcontextprotocol/go-sdk/mcp"
	x402 "github.com/x402-foundation/x402/go/v2"
	"github.com/x402-foundation/x402/go/v2/types"
)

type nativeResultCaller func(context.Context, *mcp.CallToolParams) (*mcp.CallToolResult, error)

func (f nativeResultCaller) CallTool(ctx context.Context, params *mcp.CallToolParams) (*mcp.CallToolResult, error) {
	return f(ctx, params)
}

func TestX402MCPClient_PreservesRawResult(t *testing.T) {
	for _, tc := range []struct {
		name      string
		paid      bool
		direct    bool
		imageOnly bool
		toolError bool
		recover   bool
	}{
		{name: "free mixed content"},
		{name: "automatic payment mixed content", paid: true},
		{name: "automatic payment image only", paid: true, imageOnly: true},
		{name: "explicit payment", paid: true, direct: true},
		{name: "nonpayment tool error", toolError: true},
		{name: "recovered payment returns final result", paid: true, direct: true, recover: true},
	} {
		t.Run(tc.name, func(t *testing.T) {
			native := &mcp.CallToolResult{
				Content: []mcp.Content{
					&mcp.TextContent{
						Text:        `{"answer":42}`,
						Meta:        mcp.Meta{"fixture/text": "metadata"},
						Annotations: &mcp.Annotations{Priority: 0.7},
					},
					&mcp.ImageContent{Data: []byte{1, 2, 3}, MIMEType: "image/png"},
					&mcp.AudioContent{Data: []byte{4, 5, 6}, MIMEType: "audio/wav"},
					&mcp.ResourceLink{URI: "https://example.invalid/result", Name: "result"},
					&mcp.EmbeddedResource{Resource: &mcp.ResourceContents{URI: "fixture://result", Text: "embedded result"}},
				},
				StructuredContent: map[string]interface{}{"answer": 42},
				Meta:              mcp.Meta{"fixture/vendor": "metadata"},
				IsError:           tc.toolError,
			}
			if tc.imageOnly {
				native.Content = native.Content[1:2]
				native.StructuredContent = nil
			}
			if tc.paid {
				native.Meta[MCP_PAYMENT_RESPONSE_META_KEY] = x402.SettleResponse{
					Success: true, Transaction: "fixture-transaction", Network: "eip155:1",
				}
			}
			before, err := json.Marshal(native)
			if err != nil {
				t.Fatal(err)
			}

			requirements := testPaymentRequirements()
			challenge := mcpToolResultToCallToolResult(paymentRequiredMCPToolResult(t, types.PaymentRequired{
				X402Version: 2,
				Accepts:     []types.PaymentRequirements{requirements},
			}))
			calls := 0
			caller := nativeResultCaller(func(_ context.Context, params *mcp.CallToolParams) (*mcp.CallToolResult, error) {
				calls++
				if calls == 1 && ((tc.paid && !tc.direct) || tc.recover) {
					return challenge, nil
				}
				if (params.Meta[MCP_PAYMENT_META_KEY] != nil) != tc.paid {
					t.Fatal("unexpected payment payload presence")
				}
				return native, nil
			})
			scheme := &hookSchemeMCPClient{scheme: requirements.Scheme, signalRecover: tc.recover}
			paymentClient := x402.Newx402Client().DisableSpendControls().Register(x402.Network(requirements.Network), scheme)
			client := NewX402MCPClient(caller, paymentClient, Options{})
			var result *MCPToolCallResult
			if tc.direct {
				result, err = client.CallToolWithPayment(context.Background(), "fixture", nil, types.PaymentPayload{
					X402Version: 2, Accepted: requirements, Payload: map[string]interface{}{"fixture": true},
				})
			} else {
				result, err = client.CallTool(context.Background(), "fixture", nil)
			}
			if err != nil {
				t.Fatal(err)
			}
			if result.RawResult != native {
				t.Fatal("RawResult must expose the terminal SDK result, not a projection or payment challenge")
			}
			after, err := json.Marshal(result.RawResult)
			if err != nil || string(after) != string(before) {
				t.Fatalf("native result changed: %s (error: %v)", after, err)
			}
			if tc.imageOnly {
				if len(result.Content) != 0 {
					t.Fatal("image-only result changed the text projection")
				}
			} else if len(result.Content) != 1 || result.Content[0].Type != "text" || result.Content[0].Text != `{"answer":42}` {
				t.Fatalf("text projection changed: %+v", result.Content)
			}
			if result.IsError != tc.toolError || result.PaymentMade != tc.paid {
				t.Fatalf("result flags changed: %+v", result)
			}
			if tc.paid {
				if result.PaymentResponse == nil || !result.PaymentResponse.Success || result.PaymentResponse.Transaction != "fixture-transaction" || result.PaymentResponse.Network != x402.Network(requirements.Network) {
					t.Fatalf("payment receipt changed: %+v", result.PaymentResponse)
				}
			} else if result.PaymentResponse != nil {
				t.Fatal("unexpected payment receipt")
			}
			wantCalls := 1
			if (tc.paid && !tc.direct) || tc.recover {
				wantCalls = 2
			}
			if calls != wantCalls {
				t.Fatalf("got %d calls, want %d", calls, wantCalls)
			}
		})
	}
}


func TestFinalAudit_NativeOwnershipTypedStructuredAndWrapperJSON(t *testing.T) {
 type typedOutput struct { Answer int `json:"answer"` }
 structured := &typedOutput{Answer: 42}
 text := &mcp.TextContent{Text: "original", Meta: mcp.Meta{"audit/content": "meta"}, Annotations: &mcp.Annotations{Priority: 0.8}}
 receipt := &x402.SettleResponse{Success: true, Transaction: "audit-original", Network: "eip155:1"}
 native := &mcp.CallToolResult{
  Content: []mcp.Content{
   text,
   &mcp.ImageContent{Data: []byte{7, 8}, MIMEType: "image/png"},
   &mcp.AudioContent{Data: []byte{9, 10}, MIMEType: "audio/wav"},
   &mcp.ResourceLink{URI: "https://audit.invalid/result", Name: "audit", MIMEType: "application/json"},
   &mcp.EmbeddedResource{Resource: &mcp.ResourceContents{URI: "audit://result", MIMEType: "text/plain", Text: "embedded"}},
  },
  StructuredContent: structured,
  Meta: mcp.Meta{MCP_PAYMENT_RESPONSE_META_KEY: receipt, "audit/vendor": map[string]interface{}{"retained": true}},
 }
 requirements := testPaymentRequirements()
 client := NewX402MCPClient(nativeResultCaller(func(context.Context, *mcp.CallToolParams) (*mcp.CallToolResult, error) { return native, nil }),
  x402.Newx402Client().DisableSpendControls().Register(x402.Network(requirements.Network), &hookSchemeMCPClient{scheme: requirements.Scheme}), Options{})
 result, err := client.CallToolWithPayment(context.Background(), "audit", nil, types.PaymentPayload{X402Version: 2, Accepted: requirements, Payload: map[string]interface{}{"audit": true}})
 if err != nil { t.Fatal(err) }
 if result.RawResult != native || result.RawResult.StructuredContent != structured { t.Fatal("typed structured content or native pointer changed") }
 encoded, err := json.Marshal(result)
 if err != nil { t.Fatal(err) }
 var wrapper map[string]json.RawMessage
 if err := json.Unmarshal(encoded, &wrapper); err != nil { t.Fatal(err) }
 for _, key := range []string{"Content", "IsError", "PaymentResponse", "PaymentMade", "RawResult"} {
  if _, present := wrapper[key]; !present { t.Fatalf("missing default wrapper JSON key %q", key) }
 }
 if len(wrapper) != 5 || string(wrapper["PaymentMade"]) != "true" || string(wrapper["IsError"]) != "false" { t.Fatalf("wrapper JSON shape changed: %s", encoded) }
 var projection []MCPContentItem
 if err := json.Unmarshal(wrapper["Content"], &projection); err != nil || len(projection) != 1 || projection[0].Type != "text" || projection[0].Text != "original" { t.Fatalf("bad text projection in JSON: %s, %v", encoded, err) }
 var roundtrip mcp.CallToolResult
 if err := json.Unmarshal(wrapper["RawResult"], &roundtrip); err != nil { t.Fatal(err) }
 if len(roundtrip.Content) != 5 || roundtrip.StructuredContent.(map[string]interface{})["answer"] != float64(42) || roundtrip.Meta["audit/vendor"].(map[string]interface{})["retained"] != true { t.Fatalf("native payload lost on wrapper JSON roundtrip: %s", encoded) }
 rtText, ok := roundtrip.Content[0].(*mcp.TextContent)
 if !ok || rtText.Meta["audit/content"] != "meta" || rtText.Annotations == nil || rtText.Annotations.Priority != 0.8 { t.Fatal("text annotations or metadata lost") }
 image, ok := roundtrip.Content[1].(*mcp.ImageContent)
 if !ok || image.MIMEType != "image/png" || string(image.Data) != string([]byte{7,8}) { t.Fatal("image bytes or MIME lost") }
 audio, ok := roundtrip.Content[2].(*mcp.AudioContent)
 if !ok || audio.MIMEType != "audio/wav" || string(audio.Data) != string([]byte{9,10}) { t.Fatal("audio bytes or MIME lost") }
 link, ok := roundtrip.Content[3].(*mcp.ResourceLink)
 if !ok || link.URI != "https://audit.invalid/result" || link.Name != "audit" || link.MIMEType != "application/json" { t.Fatal("resource link fields lost") }
 embedded, ok := roundtrip.Content[4].(*mcp.EmbeddedResource)
 if !ok || embedded.Resource.URI != "audit://result" || embedded.Resource.Text != "embedded" || embedded.Resource.MIMEType != "text/plain" { t.Fatal("embedded resource fields lost") }
 var jsonReceipt x402.SettleResponse
 if err := json.Unmarshal(wrapper["PaymentResponse"], &jsonReceipt); err != nil || !jsonReceipt.Success || jsonReceipt.Transaction != "audit-original" || jsonReceipt.Network != "eip155:1" { t.Fatalf("receipt changed in JSON: %s (%v)", encoded, err) }
 text.Text = "mutated"
 structured.Answer = 43
 native.IsError = true
 receipt.Transaction = "audit-mutated"
 if result.RawResult.Content[0].(*mcp.TextContent).Text != "mutated" || result.RawResult.StructuredContent.(*typedOutput).Answer != 43 || !result.RawResult.IsError { t.Fatal("RawResult does not share native ownership") }
 if result.Content[0].Text != "original" || result.IsError || result.PaymentResponse.Transaction != "audit-original" { t.Fatal("compatibility projection or extracted receipt did not retain original values") }
 zeroJSON, err := json.Marshal(MCPToolCallResult{})
 if err != nil { t.Fatal(err) }
 var zero map[string]json.RawMessage
 if err := json.Unmarshal(zeroJSON, &zero); err != nil || string(zero["RawResult"]) != "null" { t.Fatalf("unexpected zero-value RawResult JSON: %s (%v)", zeroJSON, err) }
}
